import React from 'react';

export const Card = ({ children, href, icon, target, title }) => {
  const Tag = href ? 'a' : 'div';
  const rel = target === '_blank' ? 'noopener noreferrer' : undefined;

  return (
    <Tag
      className={`
        rounded-md border border-gray-200 bg-white shadow-sm p-5 pb-0 dark:bg-inherit dark:border-white/10
        ${href ? 'no-underline! text-inherit! hover:bg-gray-100 dark:hover:bg-white/10' : ''}
      `}
      href={href}
      target={target}
      rel={rel}
    >
      {icon && (
        <img
          src={icon}
          alt=""
          className="mb-3"
          style={{ width: '32px', height: '32px' }}
        />
      )}
      {title && (
        <p className="font-semibold mb-2" style={{ color: '#5745C6' }}>
          {title}
        </p>
      )}
      <div className="text-sm leading-6 font-normal text-gray-600 dark:text-gray-300">
        {children}
      </div>
    </Tag>
  );
};

export const Cards = ({ columns = 2, children }) => {
  // Використовуємо responsive-класи замість динамічного рядка
  const gridClasses = {
    1: 'grid-cols-1',
    2: 'grid-cols-1 md:grid-cols-2',
    3: 'grid-cols-1 md:grid-cols-2 lg:grid-cols-3',
    4: 'grid-cols-1 md:grid-cols-2 lg:grid-cols-4',
  };

  return (
    <div className={`grid gap-5 ${gridClasses[columns] || 'grid-cols-1 md:grid-cols-2'}`}>
      {children}
    </div>
  );
};
